cmake_minimum_required(VERSION 3.20)
project(licensebox_cli VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
find_path(LICENSEBOX_INCLUDE_DIR licensebox_json.h HINTS "$ENV{LICENSEBOX_LIB_DIR}/../include" REQUIRED)
find_library(LICENSEBOX_LIBRARY licensebox HINTS "$ENV{LICENSEBOX_LIB_DIR}" REQUIRED)
find_path(JSON_C_INCLUDE_DIR json-c/json.h REQUIRED)
find_library(JSON_C_LIBRARY json-c REQUIRED)
add_library(cli_dispatch STATIC src/dispatch.c)
target_include_directories(cli_dispatch PUBLIC src "${LICENSEBOX_INCLUDE_DIR}")
target_link_libraries(cli_dispatch PUBLIC "${LICENSEBOX_LIBRARY}")
add_executable(licensebox src/main.c)
target_include_directories(licensebox PRIVATE "${JSON_C_INCLUDE_DIR}")
target_link_libraries(licensebox PRIVATE cli_dispatch "${JSON_C_LIBRARY}" m)
target_compile_options(licensebox PRIVATE -Wall -Wextra -Werror -Wno-misleading-indentation)
include(CTest)
if(BUILD_TESTING)
 add_executable(adapter tests/adapter.c)
 target_include_directories(adapter PRIVATE "${JSON_C_INCLUDE_DIR}")
 target_link_libraries(adapter PRIVATE cli_dispatch "${JSON_C_LIBRARY}")
endif()
install(TARGETS licensebox RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licensebox-cli)
